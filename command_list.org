#+TITLE: A Collection of Useful Local Commands

* Copying Files to Test Machines

** HP X2 210 Tablet PC

#+begin_src bash
  scp ~/src/ltc-timecode-generator/target/release/ltc-gui hp-x2:/home/viktoria/Documents/ltc-gui
  scp ~/src/ltc-timecode-generator/target/release/ltc-slint hp-x2:/home/viktoria/Documents/ltc-slint
  # In an ssh window
  sudo cp /home/viktoria/Documents/ltc-gui /home/agere/Desktop/ltc-gui
  sudo cp /home/viktoria/Documents/ltc-slint /home/agere/Desktop/ltc-slint
#+end_src

Copy logs to dev pc:
#+begin_src bash
  # In an ssh window
  sudo cp /home/agere/Documents/ltc-gen.01.log /home/viktoria/Documents/ltc-gen.01.log
  # In normal bash
  scp hp-x2:/home/viktoria/Documents/ltc-gen.01.log /home/viktoria/src/ltc-timecode-generator/logs/
#+end_src
